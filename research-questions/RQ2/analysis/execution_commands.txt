1. Get the enemy binary onto the node (once):
cd ~/GraduationProject/research-questions/RQ2
make -C stress
kubectl -n rq2-node-prep-worker6 exec -i "$AGENT" -- \
    nsenter --target 1 --mount -- bash -c 'cat > /usr/local/bin/rq2-enemy && chmod +x /usr/local/bin/rq2-enemy' \
    < stress/enemy

2. Cache condition:
# start cache enemy, detached, pidfile on the host
kubectl -n rq2-node-prep-worker6 exec -i "$AGENT" -- nsenter --target 1 --mount --pid -- bash -c \
    'nohup /usr/local/bin/rq2-enemy --size-kb 266240 --stride-bytes 64 --mode rw --cpu 2 >/tmp/rq2-enemy.log 2>&1 & echo $! > /tmp/rq2-enemy.pid; disown'

kubectl apply -f workload/pods/single_core_pod.yaml
workload/pull_results.sh single_core cache

# stop it
kubectl -n rq2-node-prep-worker6 exec -i "$AGENT" -- nsenter --target 1 --mount --pid -- bash -c \
    'kill "$(cat /tmp/rq2-enemy.pid)" && rm -f /tmp/rq2-enemy.pid'

3. Memory condition (same shape, swap size and label):
kubectl -n rq2-node-prep-worker6 exec -i "$AGENT" -- nsenter --target 1 --mount --pid -- bash -c \
    'nohup /usr/local/bin/rq2-enemy --size-kb 2662400 --stride-bytes 64 --mode rw --cpu 2 >/tmp/rq2-enemy.log 2>&1 & echo $! > /tmp/rq2-enemy.pid; disown'
